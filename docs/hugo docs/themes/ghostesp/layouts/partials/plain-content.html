{{- /* Converts a page's rendered HTML into clean, readable plain text. Preserves markdown heading markers, lists, tables, and link targets. Context: a Page. Returns a string. */ -}} {{- $page := . -}} {{- $html := $page.Content -}} {{- $html = replaceRE `(?s)]*>.*?` "" $html -}} {{- $html = replaceRE `(?s)]*href="([^"]*)"[^>]*>(.*?)` "[${2}](${1})" $html -}} {{- $html = replaceRE `(?s)]*>\s*(.*?)\s*` "\n# ${1}\n" $html -}} {{- $html = replaceRE `(?s)]*>\s*(.*?)\s*` "\n## ${1}\n" $html -}} {{- $html = replaceRE `(?s)]*>\s*(.*?)\s*` "\n### ${1}\n" $html -}} {{- $html = replaceRE `(?s)]*>\s*(.*?)\s*` "\n#### ${1}\n" $html -}} {{- $html = replaceRE `(?s)]*>\s*(.*?)\s*` "\n##### ${1}\n" $html -}} {{- $html = replaceRE `(?s)]*>\s*(.*?)\s*` "\n###### ${1}\n" $html -}} {{- $html = replaceRE `(?s)]*>(.*?)` "\n- ${1}" $html -}} {{- $html = replaceRE `(?s)]*>(.*?)` "${1} | " $html -}} {{- $html = replaceRE `(?s)` "\n" $html -}} {{- $html = replaceRE `(?s)` "\n\n" $html -}} {{- $html = replaceRE `(?s)<[^>]+>` "" $html -}} {{- $body := $html | htmlUnescape -}} {{- $body = replaceRE `(?m)[ \t]+` " " $body -}} {{- $body = replaceRE `[ \t]+\n` "\n" $body -}} {{- $body = replaceRE `(?m)^\s+` "" $body -}} {{- $body = replaceRE `\n{3,}` "\n\n" $body -}} {{- return (strings.TrimSpace $body) -}}
