{{- /* Sequence / packet-flow diagram. DSL: participant Client (optional; sets column order) Client -> AP : label | detail shown on hover Separate multiple detail lines with ";;". Arrow direction follows the from/to order, so write the pair either way. */ -}} {{- $title := .Get "title" -}} {{- $caption := .Get "caption" -}} {{- $lines := split (.Inner | strings.TrimSpace) "\n" -}} {{- $participants := slice -}} {{- $messages := slice -}} {{- range $lines -}} {{- $line := strings.TrimSpace . -}} {{- if $line -}} {{- $p := split $line " " -}} {{- if eq (index $p 0) "participant" -}} {{- $name := strings.TrimSpace (delimit (after 1 $p) " ") -}} {{- if not (in $participants $name) -}}{{- $participants = $participants | append $name -}}{{- end -}} {{- else -}} {{- $parts := split $line "->" -}} {{- if eq (len $parts) 2 -}} {{- $from := strings.TrimSpace (index $parts 0) -}} {{- $rest := strings.TrimSpace (index $parts 1) -}} {{- $tip := "" -}} {{- if in $rest "|" -}} {{- $rps := split $rest "|" -}} {{- $rest = strings.TrimSpace (index $rps 0) -}} {{- $tip = strings.TrimSpace (delimit (after 1 $rps) "|") -}} {{- end -}} {{- $to := $rest -}} {{- $label := "" -}} {{- if in $rest ":" -}} {{- $rp := split $rest ":" -}} {{- $to = strings.TrimSpace (index $rp 0) -}} {{- $label = strings.TrimSpace (delimit (after 1 $rp) ":") -}} {{- end -}} {{- if not (in $participants $from) -}}{{- $participants = $participants | append $from -}}{{- end -}} {{- if not (in $participants $to) -}}{{- $participants = $participants | append $to -}}{{- end -}} {{- $messages = $messages | append (dict "from" $from "to" $to "label" $label "tip" $tip) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $n := len $participants -}} {{- $idx := dict -}} {{- range $i, $name := $participants -}}{{- $idx = merge $idx (dict $name $i) -}}{{- end -}}
